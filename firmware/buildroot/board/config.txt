# Boot configuration for QDOS on a Raspberry Pi Zero 2 W.
#
# A calculator is judged on how quickly it is usable, so this trims what the
# firmware does before the kernel runs. The GPU stage is still about a second
# and cannot be removed.

start_file=start.elf
fixup_file=fixup.dat

kernel=Image
arm_64bit=1

# Nothing here drives a display through the GPU, so give it the minimum
gpu_mem=16

# Skip the rainbow splash and the boot delay
disable_splash=1
boot_delay=0

disable_overscan=1

# Serial console on ttyAMA0 for bring-up
dtoverlay=miniuart-bt
enable_uart=1

# --- Display -----------------------------------------------------------------
# HDMI works out of the box and is right for bring-up. An SPI panel needs its
# own overlay; uncomment and adjust for the panel actually fitted, then point
# QDOS at the node it creates with QDOS_FB.
#
# dtoverlay=spi0-1cs
# dtoverlay=fbtft,spi0-0,st7789v,width=240,height=320,rotate=90,bgr=1,speed=40000000
# dtparam=dc_pin=24,reset_pin=25

# --- Keypad ------------------------------------------------------------------
# A GPIO matrix becomes an evdev node through the matrix-keypad overlay, so no
# userspace scanning is needed. Adjust the pins to the wiring actually used.
#
# dtoverlay=matrix-keypad,rows=5,cols=5
